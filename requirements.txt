# SabzalStudio dependencies.
#
# Python 3.10 - 3.12. Not 3.13: torch 2.3.1 and numpy 1.26.4 publish no wheels
# for it, so `pip install` fails outright rather than degrading.
#
# Only direct dependencies are pinned. Transitive ones (aiosignal, frozenlist,
# h11, httpcore, multidict, yarl and the rest) were pinned here previously,
# which froze pip's resolver and made every upgrade a conflict for no benefit.

# --- Interface -------------------------------------------------------
streamlit>=1.40

# --- Language models and HTTP ----------------------------------------
requests>=2.32.3
aiohttp>=3.9.5            # async trend source fetching

# --- Article extraction ----------------------------------------------
# The "From a link" tab's main readability extractor. Without it the tab
# silently falls back to a crude tag-stripper and loses most article bodies.
trafilatura>=1.12.0

# --- Speech ----------------------------------------------------------
# 6.x sends an outdated Sec-MS-GEC-Version and Microsoft answers HTTP 403.
edge-tts>=7.2.3
# Lets EdgeTTS run inside a notebook, where a loop is already active.
nest-asyncio>=1.6.0

# --- Transcription ---------------------------------------------------
# 20231117 and 20240930 ship a setup.py that imports pkg_resources, which
# setuptools 81+ no longer provides, so `pip install` fails while building the
# wheel. 20250625 dropped that import and installs cleanly.
openai-whisper>=20250625
whisper-timestamped==1.15.4
Cython>=3.0.10            # build dependency of whisper-timestamped
dtw-python>=1.5.1         # whisper-timestamped's alignment backend
torch==2.3.1              # no cp313 wheel; see the Python note above
numpy>=1.26.4,<2          # whisper and numba are not numpy 2 ready

# --- Video and images ------------------------------------------------
moviepy==1.0.3
# MoviePy 1.0.3 resizes through PIL constants that Pillow 10 removed.
# utility/core/pillow_compat.py restores them, so Pillow 10 is safe here.
pillow>=10.3.0
imageio>=2.34.1
imageio-ffmpeg>=0.5.1     # bundles the ffmpeg binary; no separate install
