# SabzalStudio dependencies.
#
# Python 3.10 - 3.12.
#
# Nothing here is pinned to an exact version unless a real incompatibility
# demands it. An over-tight pin is not "safe": on a host that already ships a
# working build (Colab, Kaggle, a GPU image) pip will tear it out and install
# a worse one, and downgrading a compiled package over an existing install can
# leave mismatched .so files behind.
#
# Only direct dependencies are pinned. Transitive ones (aiosignal, frozenlist,
# h11, httpcore, multidict, yarl and the rest) were pinned here previously,
# which froze pip's resolver and made every upgrade a conflict for no benefit.

# --- Interface -------------------------------------------------------
streamlit>=1.40

# --- Language models and HTTP ----------------------------------------
requests>=2.32.3
aiohttp>=3.9.5            # async trend source fetching

# --- Article extraction ----------------------------------------------
# The "From a link" tab's main readability extractor. Without it the tab
# silently falls back to a crude tag-stripper and loses most article bodies.
trafilatura>=1.12.0

# --- Speech ----------------------------------------------------------
# 6.x sends an outdated Sec-MS-GEC-Version and Microsoft answers HTTP 403.
edge-tts>=7.2.3
# Lets EdgeTTS run inside a notebook, where a loop is already active.
nest-asyncio>=1.6.0

# --- Transcription ---------------------------------------------------
# 20231117 and 20240930 ship a setup.py that imports pkg_resources, which
# setuptools 81+ no longer provides, so `pip install` fails while building the
# wheel. 20250625 dropped that import and installs cleanly.
openai-whisper>=20250625
whisper-timestamped==1.15.4
Cython>=3.0.10            # build dependency of whisper-timestamped
dtw-python>=1.5.1         # whisper-timestamped's alignment backend
# openai-whisper asks for a bare "torch" with no bound, so there is no reason
# to pin one. Pinning 2.3.1 was actively harmful on Colab and similar hosts,
# which ship a torch built against their own CUDA: pip would replace a working
# GPU build with an ~800 MB CPU one. Any torch >= 2 works.
torch>=2.0
# Bounded by numba, which whisper pulls in, not by whisper itself. numba 0.60
# takes numpy<2.1, 0.61 takes <2.2, 0.62 takes <2.4 -- so pip resolves this
# against whichever numba it picks. Pinning numpy<2 outright was wrong: it
# forced a downgrade on hosts that already ship a compatible numpy 2.0, and a
# downgrade over an existing install leaves stale compiled .so files behind,
# which surfaces as "numpy.dtype size changed" on the next import.
numpy>=1.26.4

# --- Video and images ------------------------------------------------
moviepy==1.0.3
# MoviePy 1.0.3 resizes through PIL constants that Pillow 10 removed.
# utility/core/pillow_compat.py restores them, so Pillow 10 is safe here.
pillow>=10.3.0
imageio>=2.34.1
imageio-ffmpeg>=0.5.1     # bundles the ffmpeg binary; no separate install
